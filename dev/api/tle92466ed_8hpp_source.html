351[[nodiscard]] constexpr Result<ChannelFeedback> DecodeFeedback(const std::array<uint32_t, 5>& w,
352 uint32_t system_hz) noexcept {
353 const uint32_t tp = reg::TpMant::Extract(w[0]);
354 if ((tp == 0U) || (system_hz == 0U)) {
355 return Result<ChannelFeedback>::Failure(Error::NotReady);
356 }
357 const uint32_t to = reg::ToMant::Extract(w[0]);
358 ChannelFeedback f{};
359 f.average_current_ua = static_cast<int32_t>(
360 (4'000'000 * static_cast<int64_t>(SignExtend<12U>(reg::IAvgMant::Extract(w[2])))) /
361 static_cast<int64_t>(tp));
362 const uint32_t duty = (to * 10000U) / tp;
363 f.duty_permyriad =
364 (duty < 10000U) ? duty : 10000U; // TO_MANT never exceeds TP_MANT on a real chip
365 f.vbat_mv = (reg::VbatAvgMant::Extract(w[1]) * 41470U) / tp;
366 f.period_us = static_cast<uint32_t>(
367 ((static_cast<uint64_t>(tp) << reg::FbExp::Extract(w[1])) * 1'000'000U) / system_hz);
368 f.min_current_ua = (4'000'000 * SignExtend<10U>(reg::IMin::Extract(w[3]))) / 511;
369 f.max_current_ua = (4'000'000 * SignExtend<10U>(reg::IMax::Extract(w[3]))) / 511;
370 f.min_pwm_period_ns = static_cast<uint32_t>(
371 (static_cast<uint64_t>(reg::PMin::Extract(w[4])) * 256'000'000'000U) / system_hz);
372 f.max_pwm_period_ns = static_cast<uint32_t>(
373 (static_cast<uint64_t>(reg::PMax::Extract(w[4])) * 256'000'000'000U) / system_hz);
374 f.period_count = static_cast<uint8_t>(reg::DitherPeriodCnt::Extract(w[1]));
375 f.quad_count = static_cast<uint8_t>(reg::DitherQuadCnt::Extract(w[1]));
376 return Result<ChannelFeedback>::Success(f);
377}
