369[[nodiscard]] constexpr uint8_t FrameCrc(uint32_t frame) noexcept {
370 uint32_t crc = kCrcInit; // all arithmetic in uint32_t (KB-CPP20-021)
371 for (uint32_t byte = 0U; byte < 3U; ++byte) {
372 crc ^= (frame >> (8U * byte)) & 0xFFU;
373 for (uint32_t bit = 0U; bit < 8U; ++bit) {
374 crc = ((crc & 0x80U) != 0U) ? ((crc << 1U) ^ kCrcPolynomial) : (crc << 1U);
375 crc &= 0xFFU;
376 }
377 }
378 return static_cast<uint8_t>(crc ^ kCrcXorOut);
379}

394[[nodiscard]] constexpr bool Is22Bit(uint16_t address) noexcept {

409[[nodiscard]] constexpr bool IsRegister(uint16_t address) noexcept {
410 const auto a = static_cast<uint32_t>(address);
411 // CH_CTRL to CLK_DIV, one bit per listed address of Table 19.
412 constexpr uint32_t kCentral = 0x03C7'1FFDU;
413 if (a <= 0x0019U) {
414 return ((kCentral >> a) & 1U) != 0U;
415 }
416 if ((a == kSffBist) || ((a >= kIcvid) && (a <= 0x0207U))) {
417 return true;
418 }
419 const uint32_t bank = a & 0xFDF0U; // the channel base, from either bank
420 const uint32_t offset = a & 0x000FU;
421 const bool is_base = (bank >= 0x0020U) && (bank <= 0x0070U);
422 const bool control =
423 (a < 0x0200U) && ((offset <= 0x7U) || ((offset >= 0xCU) && (offset <= 0xEU)));
424 const bool feedback = (a >= 0x0200U) && (offset <= 0x6U);
425 return is_base && (control || feedback);
426}
427static_assert(IsRegister(kChCtrl) && !IsRegister(0x0001U) && IsRegister(kSffBist));
428static_assert(IsRegister(0x0040U) && IsRegister(0x004EU) && !IsRegister(0x0048U));
429static_assert(IsRegister(0x0246U) && !IsRegister(0x0247U) && !IsRegister(0x0010U + 3U));
